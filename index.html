const AUDIO_URL = "https://www.image2url.com/r2/default/audio/1795119985931-bfc559b7-1f06-4403-8735-f6deff3510d4.mp3"; const audio = new Audio(AUDIO_URL); audio.preload = 'auto'; const statusBox = document.getElementById('status'); const params = new URLSearchParams(window.location.search); const currentId = params.get('id'); const previousId = sessionStorage.getItem('ultimo_event_id'); // All'avvio, controlla se il browser permette già l'autoplay o se serve il clic window.addEventListener('DOMContentLoaded', () => { if (currentId && previousId && currentId !== previousId) { // Prova a suonare per il nuovo terremoto audio.play().catch(() => { statusBox.className = 'visibile'; }); } else { // Test silenzioso al caricamento per capire se l'audio è sbloccato audio.muted = true; audio.play().then(() => { audio.pause(); audio.currentTime = 0; audio.muted = false; }).catch(() => { audio.muted = false; statusBox.className = 'visibile'; // Mostra subito l'avviso se il browser ha bloccato il dominio }); } if (currentId) { sessionStorage.setItem('ultimo_event_id', currentId); } }); // Funzione di sblocco al click function sbloccaAudio() { audio.muted = false; audio.play().then(() => { // Lascia suonare per 100ms prima di fermare, così il browser registra l'interazione setTimeout(() => { audio.pause(); audio.currentTime = 0; }, 100); statusBox.className = ''; }).catch(e => { console.error("Errore sblocco:", e); }); }
